Throttling Strategy
Two mechanisms prevent queue buildup:

Busy gate (useObjectDetection.ts): isInferring SharedValue — if a previous runAsync dispatch hasn't completed, the frame is dropped immediately. This is the primary backpressure mechanism.

Time throttle (useObjectDetection.ts): 300ms minimum interval between dispatches. Even if the busy gate is free, frames arriving sooner than 300ms after the last dispatch are dropped.

Combined, these guarantee effective inference FPS of at most ~3.3 FPS (1000ms / 300ms), independent of actual inference speed. The code comment estimates ~3–5 FPS effective on Snapdragon 695.

Detection → React Bridge
Results return to the JS thread via Worklets.createRunOnJS() at useObjectDetection.ts. The callback chain is:

runOnJs(name) → handleDetectionCallback(name) → setLabel(name) + speakDetection(name)
setLabel updates React state (triggers re-render of the <Text> overlay). speakDetection calls Speech.speak() with a 3-second cooldown (separate from the inference throttle) to prevent TTS spam.

class BoundingBoxPainter extends CustomPainter {
  final List<Detection> detections;
  final Size imageSize; // 320x320

  BoundingBoxPainter(this.detections, this.imageSize);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.red
      ..style = PaintingStyle.stroke
      ..strokeWidth = 3.0;

    for (var det in detections) {
      // Scale normalized coords to screen size
      final rect = Rect.fromLTWH(
        det.box.left * size.width,
        det.box.top * size.height,
        det.box.width * size.width,
        det.box.height * size.height,
      );
      canvas.drawRect(rect, paint);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}

class BoundingBoxOverlay extends CustomPainter {
  final List<Detection> detections;

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.red
      ..style = PaintingStyle.stroke
      ..strokeWidth = 3.0;

    for (var det in detections) {
      // Scale normalized coords (0.0-1.0) to screen size
      final rect = Rect.fromLTWH(
        det.x * size.width,
        det.y * size.height,
        det.w * size.width,
        det.h * size.height,
      );
      canvas.drawRect(rect, paint);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;

final FlutterTts tts = FlutterTts();
String _lastSpokenObject = "";
DateTime _lastSpokenTime = DateTime.now();

void handleAudioFeedback(List<Detection> detections) {
  if (detections.isEmpty) return;
  
  // Sort by confidence or predefined priority (e.g., Person > Chair)
  detections.sort((a, b) => b.confidence.compareTo(a.confidence));
  var topDetection = detections.first;
  
  bool isRecent = DateTime.now().difference(_lastSpokenTime).inSeconds < 3;
  
  // Only speak if it's a new object, or enough time has passed
  if (topDetection.label != _lastSpokenObject || !isRecent) {
    tts.speak("${topDetection.label} ahead");
    _lastSpokenObject = topDetection.label;
    _lastSpokenTime = DateTime.now();
  }
}

class InferenceService {
  late Isolate _isolate;
  late SendPort _sendPort;
  late ReceivePort _receivePort;

  // Stream to feed results back to the UI
  late StreamController<List<Detection>> _resultsController;
  Stream<List<Detection>> get resultsStream => _resultsController.stream;

  Future<void> initIsolate() async {
    _resultsController = StreamController.broadcast();
    _receivePort = ReceivePort();
    
    // Spawn the isolate ONCE
    _isolate = await Isolate.spawn(
      _isolateEntryPoint,
      _receivePort.sendPort,
    );

    // Listen for messages from the isolate
    _receivePort.listen((dynamic message) {
      if (message is SendPort) {
        _sendPort = message; // Isolate is ready
      } else if (message is List<Detection>) {
        _resultsController.add(message); // Inference results
      }
    });
  }

  void processFrame({
    required Uint8List yPlane,
    required Uint8List uPlane,
    required Uint8List vPlane,
    required int width,
    required int height,

    // Send raw bytes to the isolate
    _sendPort.send({
      'y': yPlane,
      'u': uPlane,
      'v': vPlane,
      'width': width,
      'height': height,
    });
  }

    _isolate.kill(priority: Isolate.immediate);
    _resultsController.close();

  static void _isolateEntryPoint(SendPort mainSendPort) {
    final isolateReceivePort = ReceivePort();
    mainSendPort.send(isolateReceivePort.sendPort);

    // 1. Load Interpreter ONCE with XNNPACK
    final options = InterpreterOptions()..useXnnpack = true;
    final interpreter = Interpreter.fromAsset('assets/yolov8n_int8.tflite', options: options);

    // Get output tensor shape dynamically (usually [1, 84, 2100] for YOLOv8)
    final outputShape = interpreter.getOutputTensor(0).shape;
    final numClasses = outputShape[1] - 4; // Usually 80
    final numBoxes = outputShape[2]; // Usually 2100 or 8400

    // 2. Pre-allocate Buffers ONCE
    final inputBuffer = Float32List(1 * 320 * 320 * 3);
    final outputBuffer = List.filled(1 * outputShape[1] * outputShape[2], 0.0);

    // 3. Listen for frames forever
    isolateReceivePort.listen((dynamic data) {
      if (data is Map<String, dynamic>) {
        final yBytes = data['y'] as Uint8List;
        final uBytes = data['u'] as Uint8List;
        final vBytes = data['v'] as Uint8List;
        final width = data['width'] as int;
        final height = data['height'] as int;

        // --- MANUAL YUV to RGB Conversion ---
        // This bypasses the slow `image` package entirely.
        final uvRowStride = (width + 1) ~/ 2; // UV stride is half width rounded up
        final uvPixelStride = 1; // For YUV420, pixel stride is 1

        int pixelIndex = 0;
        for (int y = 0; y < height; y++) {
          for (int x = 0; x < width; x++) {
            final int uvIndex = uvPixelStride * (x ~/ 2) + uvRowStride * (y ~/ 2);

            final yp = yBytes[y * width + x];
            final up = uBytes[uvIndex];
            final vp = vBytes[uvIndex];

            // Standard YUV to RGB math
            int r = (yp + vp * 1.370705 - 179.454772).round().clamp(0, 255);
            int g = (yp - up * 0.336668 - vp * 0.698001 + 135.45984).round().clamp(0, 255);
            int b = (yp + up * 1.732446 - 226.318144).round().clamp(0, 255);

            // Normalize to [0.0, 1.0] and write to buffer
            // Note: TFLite model input is usually [1, 320, 320, 3]
            // We only write up to 320x320. If camera frame is 640x480, this crops it to top-left 320x320.
            // (For a thesis prototype, this is acceptable. Later we can add center-cropping).
            if (y < 320 && x < 320) {
              inputBuffer[pixelIndex++] = r / 255.0;
              inputBuffer[pixelIndex++] = g / 255.0;
              inputBuffer[pixelIndex++] = b / 255.0;
            }
          }
        }

        // --- INFERENCE ---
        // Reshape input for TFLite: [1, 320, 320, 3]
        final input = inputBuffer.reshape([1, 320, 320, 3]);
        interpreter.run(input, outputBuffer);

        // --- POST-PROCESSING (Argmax + NMS) ---
        List<Detection> detections = [];
        for (int i = 0; i < numBoxes; i++) {
          double maxConf = 0.0;
          int maxClassId = -1;

          // Find the highest confidence class for this box
          for (int j = 4; j < 4 + numClasses; j++) {
            final conf = outputBuffer[j * numBoxes + i];
            if (conf > maxConf) {
              maxConf = conf;
              maxClassId = j - 4;
            }
          }

          // Confidence Threshold (0.5)
          if (maxConf > 0.5) {
            // Extract normalized center coordinates [x, y, w, h]
            final cx = outputBuffer[0 * numBoxes + i];
            final cy = outputBuffer[1 * numBoxes + i];
            final w = outputBuffer[2 * numBoxes + i];
            final h = outputBuffer[3 * numBoxes + i];

            detections.add(Detection(
              classId: maxClassId,
              label: 'Object $maxClassId', // Replace with actual labels later
              confidence: maxConf,
              box: Rect.fromCenter(center: Offset(cx, cy), width: w, height: h),
            ));
          }
        }

        // Optional: Apply Non-Maximum Suppression (NMS) here to remove overlapping boxes.
        // For now, we'll just send the raw detections to the UI.

        // Send results back to main thread
        mainSendPort.send(detections);
      }
    });
  }
What this code does:
It avoids the image package completely by using raw math to convert Y, U, and V byte arrays into RGB Float32 values directly in the pre-allocated buffer.
It dynamically checks the model's output shape so it works whether you exported 2100 boxes or 8400 boxes.
It extracts the bounding box coordinates and confidence scores, filters out low-confidence boxes, and sends a list of Detection objects back to the UI.
Try hot-restarting the app (R in the terminal). Point it at an object (like a person or a chair). The app won't draw boxes on the screen yet, but if you add a print(detections.length); right before sending it back, you should see it detecting objects in the console!
